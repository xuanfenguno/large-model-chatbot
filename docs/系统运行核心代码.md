class EnhancedQwenApi:
    def __init__(self):
        self.api_key = settings.DASHSCOPE_API_KEY
        self.base_url = "https://dashscope.aliyuncs.com/api/v1"
        self.default_model = "qwen-vl-max"

    def send_message(self, message, config, image_url=None):
        if image_url:
            return self._send_multimodal(message, config, image_url)
        return self._send_text(message, config)

    def _send_text(self, message, config):
        headers = {"Authorization": f"Bearer {self.api_key}", "Content-Type": "application/json"}
        payload = {
            "model": config.get('model', self.default_model),
            "messages": [{"role": "user", "content": message}],
            "stream": False
        }
        response = requests.post(f"{self.base_url}/services/aigc/text-generation/generation", headers=headers, json=payload)
        response.raise_for_status()
        result = response.json()
        return {'content': result['output']['text'], 'usage': result.get('usage', {})}

    def _send_multimodal(self, message, config, image_url):
        headers = {"Authorization": f"Bearer {self.api_key}", "Content-Type": "application/json"}
        
        if image_url.startswith('data:'):
            image_content = image_url
        else:
            image_content = image_url

        payload = {
            "model": "qwen-vl-max",
            "input": {
                "messages": [{
                    "role": "user",
                    "content": [
                        {"image": image_content},
                        {"text": message}
                    ]
                }]
            },
            "parameters": {"result_format": "message"}
        }
        
        response = requests.post(f"{self.base_url}/services/aigc/multimodal-generation/generation", headers=headers, json=payload)
        response.raise_for_status()
        result = response.json()
        content = result['output']['choices'][0]['message']['content'][0]['text']
        return {'content': content, 'usage': result.get('usage', {})}

    def send_message_stream(self, message, config, image_url=None):
        if image_url:
            return self._send_multimodal_stream(message, config, image_url)
        return self._send_text_stream(message, config)

    def _send_text_stream(self, message, config):
        headers = {"Authorization": f"Bearer {self.api_key}", "Content-Type": "application/json"}
        payload = {
            "model": config.get('model', self.default_model),
            "messages": [{"role": "user", "content": message}],
            "stream": True
        }
        response = requests.post(f"{self.base_url}/services/aigc/text-generation/generation", headers=headers, json=payload, stream=True)
        response.raise_for_status()
        
        for line in response.iter_lines():
            if line:
                line_str = line.decode('utf-8')
                if line_str.startswith('data:'):
                    data_str = line_str[5:].strip()
                    if data_str == '[DONE]':
                        break
                    try:
                        data = json.loads(data_str)
                        if 'choices' in data and len(data['choices']) > 0:
                            delta = data['choices'][0].get('delta', {})
                            content = delta.get('content', '')
                            if content:
                                yield content
                    except json.JSONDecodeError:
                        continue

    def _send_multimodal_stream(self, message, config, image_url):
        headers = {"Authorization": f"Bearer {self.api_key}", "Content-Type": "application/json"}
        image_content = image_url if image_url.startswith('data:') else image_url
        
        payload = {
            "model": "qwen-vl-max",
            "input": {
                "messages": [{
                    "role": "user",
                    "content": [{"image": image_content}, {"text": message}]
                }]
            },
            "parameters": {"result_format": "message"}
        }
        
        response = requests.post(f"{self.base_url}/services/aigc/multimodal-generation/generation", headers=headers, json=payload)
        response.raise_for_status()
        result = response.json()
        content = result['output']['choices'][0]['message']['content'][0]['text']
        yield content

@api_view(['POST'])
@authentication_classes([JWTAuthentication])
@permission_classes([IsAuthenticated])
def chat(request):
    message = request.data.get('message', '')
    image_url = request.data.get('image_url')
    model_id = request.data.get('model', 'qwen-vl-max')
    conversation_id = request.data.get('conversation_id')

    if not message and not image_url:
        return Response({'error': '消息内容不能为空'}, status=400)

    try:
        conversation = Conversation.objects.get(id=conversation_id, user=request.user)
    except Conversation.DoesNotExist:
        conversation = Conversation.objects.create(user=request.user, title=message[:50] if message else '图片对话')

    Message.objects.create(conversation=conversation, role='user', content=message, image_url=image_url)

    api = EnhancedQwenApi()
    config = {'model': model_id}

    try:
        if image_url:
            response_data = api.send_message(message, config, image_url)
        else:
            response_data = api.send_message(message, config)

        ai_content = response_data['content']
        Message.objects.create(conversation=conversation, role='assistant', content=ai_content)

        return Response({
            'content': ai_content,
            'conversation_id': conversation.id,
            'usage': response_data.get('usage', {})
        })
    except Exception as e:
        return Response({'error': f'AI服务错误: {str(e)}'}, status=500)

def stream_chat(request):
    message = request.data.get('message', '')
    image_url = request.data.get('image_url')
    model_id = request.data.get('model', 'qwen-vl-max')
    conversation_id = request.data.get('conversation_id')

    if not message and not image_url:
        return Response({'error': '消息内容不能为空'}, status=400)

    try:
        conversation = Conversation.objects.get(id=conversation_id, user=request.user)
    except Conversation.DoesNotExist:
        conversation = Conversation.objects.create(user=request.user, title=message[:50] if message else '图片对话')

    Message.objects.create(conversation=conversation, role='user', content=message, image_url=image_url)

    api = EnhancedQwenApi()
    config = {'model': model_id}

    def event_stream():
        full_content = ""
        try:
            if image_url:
                stream = api.send_message_stream(message, config, image_url)
            else:
                stream = api.send_message_stream(message, config)

            for chunk in stream:
                if chunk:
                    full_content += chunk
                    yield f"data: {json.dumps({'content': chunk})}\n\n"

            Message.objects.create(conversation=conversation, role='assistant', content=full_content)
            yield f"data: {json.dumps({'done': True, 'conversation_id': conversation.id})}\n\n"

        except Exception as e:
            yield f"data: {json.dumps({'error': str(e)})}\n\n"

    response = StreamingHttpResponse(event_stream(), content_type='text/event-stream')
    response['Cache-Control'] = 'no-cache'
    response['X-Accel-Buffering'] = 'no'
    return response

@api_view(['POST'])
@authentication_classes([JWTAuthentication])
@permission_classes([IsAuthenticated])
def upload_image(request):
    if 'image' not in request.FILES:
        return Response({'error': '没有上传图片'}, status=400)

    image = request.FILES['image']
    
    if not image.content_type.startswith('image/'):
        return Response({'error': '只能上传图片文件'}, status=400)

    if image.size > 5 * 1024 * 1024:
        return Response({'error': '图片大小不能超过5MB'}, status=400)

    upload_path = os.path.join(settings.MEDIA_ROOT, 'chat_images')
    os.makedirs(upload_path, exist_ok=True)

    filename = f"{uuid4().hex}_{image.name}"
    file_path = os.path.join(upload_path, filename)

    with open(file_path, 'wb+') as destination:
        for chunk in image.chunks():
            destination.write(chunk)

    image_url = f"{settings.MEDIA_URL}chat_images/{filename}"
    return Response({'image_url': image_url})

export const useChatStore = defineStore('chat', {
  state: () => ({
    conversations: [],
    selectedConversationId: null,
    messages: [],
    isLoading: false,
    isStreaming: false
  }),

  actions: {
    async fetchConversations() {
      this.isLoading = true
      try {
        const response = await service.get('/conversations/')
        this.conversations = response.data
      } catch (error) {
        console.error('获取对话列表失败:', error)
        throw error
      } finally {
        this.isLoading = false
      }
    },

    async createConversation() {
      try {
        const response = await service.post('/conversations/', { title: '新对话' })
        this.conversations.unshift(response.data)
        this.selectedConversationId = response.data.id
        this.messages = []
        return response.data
      } catch (error) {
        throw error
      }
    },

    async selectConversation(id) {
      this.selectedConversationId = id
      await this.fetchMessages(id)
    },

    async fetchMessages(conversationId) {
      this.isLoading = true
      try {
        const response = await service.get(`/conversations/${conversationId}/messages/`)
        this.messages = response.data
      } catch (error) {
        throw error
      } finally {
        this.isLoading = false
      }
    },

    async sendMessage(content, imageUrl = null, model = 'qwen-vl-max') {
      if (!this.selectedConversationId) {
        await this.createConversation()
      }

      const userMessage = {
        id: Date.now(),
        role: 'user',
        content: content,
        image_url: imageUrl,
        created_at: new Date().toISOString()
      }
      this.messages.push(userMessage)

      const loadingMessage = {
        id: Date.now() + 1,
        role: 'assistant',
        content: '',
        is_loading: true,
        created_at: new Date().toISOString()
      }
      this.messages.push(loadingMessage)

      try {
        // 如果有图片，使用非流式接口
        if (imageUrl) {
          const response = await service.post('/chat/', {
            message: content,
            image_url: imageUrl,
            model: model,
            conversation_id: this.selectedConversationId
          })

          const loadingIndex = this.messages.findIndex(m => m.is_loading)
          if (loadingIndex !== -1) {
            this.messages[loadingIndex] = {
              id: loadingIndex,
              role: 'assistant',
              content: response.data.content,
              created_at: new Date().toISOString()
            }
          }

          return response.data.content
        } else {
          // 纯文字使用流式接口
          return await this.sendMessageStream(content, model, loadingMessage)
        }
      } catch (error) {
        const loadingIndex = this.messages.findIndex(m => m.is_loading)
        if (loadingIndex !== -1) {
          this.messages[loadingIndex] = {
            id: loadingIndex,
            role: 'assistant',
            content: `发送失败: ${error.message}`,
            error: true,
            created_at: new Date().toISOString()
          }
        }
        throw error
      }
    },

    async sendMessageStream(content, model, loadingMessage) {
      this.isStreaming = true
      let fullContent = ''

      try {
        const response = await fetch('/api/v1/stream-chat/', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${localStorage.getItem('token')}`
          },
          body: JSON.stringify({
            message: content,
            model: model,
            conversation_id: this.selectedConversationId
          })
        })

        const reader = response.body.getReader()
        const decoder = new TextDecoder()

        while (true) {
          const { done, value } = await reader.read()
          if (done) break

          const chunk = decoder.decode(value, { stream: true })
          const lines = chunk.split('\n')

          for (const line of lines) {
            if (line.startsWith('data: ')) {
              try {
                const data = JSON.parse(line.slice(6))
                if (data.error) {
                  throw new Error(data.error)
                }
                if (data.content) {
                  fullContent += data.content
                  const loadingIndex = this.messages.findIndex(m => m.is_loading)
                  if (loadingIndex !== -1) {
                    this.messages[loadingIndex].content = fullContent
                  }
                }
                if (data.done) {
                  const loadingIndex = this.messages.findIndex(m => m.is_loading)
                  if (loadingIndex !== -1) {
                    this.messages[loadingIndex] = {
                      id: loadingIndex,
                      role: 'assistant',
                      content: fullContent,
                      created_at: new Date().toISOString()
                    }
                  }
                  return fullContent
                }
              } catch (e) {
                if (e.message) throw e
              }
            }
          }
        }

        return fullContent
      } finally {
        this.isStreaming = false
      }
    }
  }
})

// 发送消息
const handleSendMessage = async () => {
  if ((!inputContent.value.trim() && !imagePreviewUrl.value) || isSending.value) return

  let content = inputContent.value.trim()
  inputContent.value = ''

  const tempImage = imagePreviewUrl.value ? selectedImage.value : null
  imagePreviewUrl.value = ''
  selectedImage.value = null

  isSending.value = true

  try {
    let imageUrl = null
    if (tempImage) {
      ElMessage.info('正在上传图片...')
      imageUrl = await uploadImage(tempImage)
      if (!imageUrl) throw new Error('图片上传失败')
    }

    if (!content && imageUrl) {
      content = '请详细描述这张图片的内容，包括图片中的物体、人物、场景、文字等信息。'
    }

    const response = await chatStore.sendMessage(content, imageUrl, selectedModel.value)
    
    if (chatMode.value === 'voice') await speakText(response)
  } catch (error) {
    let errorMessage = '发送消息失败'
    if (error.response) {
      const statusCode = error.response.status
      switch (statusCode) {
        case 401: errorMessage = '认证失败，请重新登录'; break
        case 403: errorMessage = '权限不足，无法发送消息'; break
        case 429: errorMessage = '请求过于频繁，请稍后再试'; break
        case 500: errorMessage = '服务器内部错误，请稍后再试'; break
        default: errorMessage = `服务器错误 (${statusCode})，请稍后再试`
      }
    } else if (error.request) errorMessage = '网络连接失败，请检查网络连接'
    else errorMessage = error.message || '发送消息时发生未知错误'
    
    ElMessage.error({ message: errorMessage, duration: 3000, showClose: true })
  } finally { isSending.value = false }
}

const uploadImage = async (imageFile) => {
  try {
    const formData = new FormData()
    formData.append('image', imageFile)
    const token = localStorage.getItem('token')
    const response = await service.post('/upload-image/', formData, {
      headers: { 'Content-Type': 'multipart/form-data', 'Authorization': `Bearer ${token}` },
      timeout: 30000
    })
    if (response.data && response.data.image_url) return response.data.image_url
    return null
  } catch (error) {
    console.error('图片上传失败:', error)
    ElMessage.error('图片上传失败，请稍后重试')
    return null
  }
}

const loadAvailableModels = async () => {
  try {
    const availableModels = await aiApi.getAvailableModels()
    const groups = {}
    availableModels.forEach(model => {
      const groupName = model.group || '通用'
      if (!groups[groupName]) groups[groupName] = []
      groups[groupName].push({ id: model.id, name: model.name, provider: model.provider, available: model.available !== undefined ? model.available : true, description: model.description })
    })
    modelGroups.value = Object.keys(groups).map(groupName => ({ label: groupName, models: groups[groupName] }))
    if (availableModels.length > 0) {
      const defaultModel = configManager.getDefaultModel()
      selectedModel.value = availableModels.some(m => m.id === defaultModel) ? defaultModel : availableModels[0].id
    }
  } catch (error) {
    console.error('加载模型列表失败:', error)
    modelGroups.value = [{ label: '模型', models: [

// 认证检查
const checkAuth = () => {
  const hasValidToken = authStore.token || localStorage.getItem('token')
  const hasValidUser = authStore.user || localStorage.getItem('user')
  const isAuthenticated = hasValidToken && hasValidUser
  if (!isAuthenticated) return false
  if (!authStore.token && localStorage.getItem('token')) {
    authStore.token = localStorage.getItem('token')
    authStore.user = JSON.parse(localStorage.getItem('user') || 'null')
  }
  return true
}

// Token自动刷新
const setupTokenRefresh = () => {
  const refreshInterval = setInterval(async () => {
    try {
      const token = localStorage.getItem('token')
      if (token && authStore.isLoggedIn) {
        await service.get('/conversations/', { timeout: 5000, _isTokenCheck: true })
      }
    } catch (error) {
      if (error.response?.status === 401) {
        try { await authStore.refreshToken() }
        catch (refreshError) { console.error('token自动刷新失败:', refreshError) }
      }
    }
  }, 30 * 60 * 1000)
  onUnmounted(() => { clearInterval(refreshInterval) })
}
