## 流式响应联调（核心代码）

```javascript
// 前端流式响应解析
const sendMessageStream = async (content) => {
  try {
    const response = await axios.post('/api/v1/chat/stream/', {
      message: content,
      model: chatStore.currentModel.value
    }, {
      headers: { 'Authorization': `Bearer ${localStorage.getItem('token')}` }
    })

    const reader = response.data.getReader()
    const decoder = new TextDecoder()
    
    while (true) {
      const { done, value } = await reader.read()
      if (done) break
      const chunk = decoder.decode(value, { stream: true })
      const lines = chunk.split('\n')
      for (const line of lines) {
        if (line.startsWith('data: ')) {
          const data = JSON.parse(line.slice(6))
          if (data.content) {
            chatStore.messages.value[chatStore.messages.value.length - 1].content += data.content
          }
        }
      }
    }
  } catch (error) {
    console.error('发送消息失败:', error)
  }
}
```

# 后端流式响应生成
@api_view(['POST'])
@permission_classes([permissions.IsAuthenticated])
def stream_chat(request):
    message_content = request.data.get('message')
    model = request.data.get('model', 'gpt-3.5-turbo')

    def stream_response_generator():
        try:
            api_instance = EnhancedApiWrapper.create_api_instance(model)
            result = api_instance.send_message(
                message=message_content,
                config={'model': model, 'history': []}
            )
            content = result['content']
            
            yield f"data: {json.dumps({'content': content})}\n\n"
            yield f"data: {json.dumps({'status': 'completed'})}\n\n"
        except Exception as e:
            logger.error(f"Streaming error: {str(e)}")
            yield f"data: {json.dumps({'error': str(e)})}\n\n"

    return StreamingHttpResponse(
        stream_response_generator(),
        content_type='text/event-stream'
    )
