```python
# backend/chatbot/views.py
from django.contrib.auth.models import User
from django.contrib.auth.hashers import make_password
from rest_framework import status
from rest_framework.response import Response
from rest_framework.decorators import api_view

@api_view(['POST'])
def register(request):
    """用户注册"""
    username = request.data.get('username')
    password = request.data.get('password')
    email = request.data.get('email', '')
    
    # 验证用户是否存在
    if User.objects.filter(username=username).exists():
        return Response(
            {'error': '用户名已存在'}, 
            status=status.HTTP_400_BAD_REQUEST
        )
    
    # 创建用户
    user = User.objects.create(
        username=username,
        password=make_password(password),
        email=email
    )
    
    # 创建用户配置
    UserProfile.objects.create(user=user)
    
    return Response(
        {'message': '注册成功'}, 
        status=status.HTTP_201_CREATED
    )
```

**实现说明**：
1. 接收用户输入的用户名、密码和邮箱
2. 验证用户名是否已存在
3. 使用 `make_password()` 对密码进行加密
4. 创建用户并自动创建用户配置
5. 返回注册结果

#### 2. 用户登录实现

```python
# backend/chatbot/views.py
from django.contrib.auth import authenticate
from rest_framework_simplejwt.tokens import RefreshToken

@api_view(['POST'])
def login(request):
    """用户登录"""
    username = request.data.get('username')
    password = request.data.get('password')
    
    # 验证用户
    user = authenticate(username=username, password=password)
    
    if user is None:
        return Response(
            {'error': '用户名或密码错误'}, 
            status=status.HTTP_401_UNAUTHORIZED
        )
    
    # 生成JWT Token
    refresh = RefreshToken.for_user(user)
    
    return Response({
        'access': str(refresh.access_token),
        'refresh': str(refresh),
        'user': {
            'id': user.id,
            'username': user.username,
            'email': user.email
        }
    })
```

**实现说明**：
1. 接收用户输入的用户名和密码
2. 使用 `authenticate()` 验证用户凭据
3. 使用 JWT 生成访问令牌和刷新令牌
4. 返回用户信息和Token

#### 2. 发送消息

```python
# backend/chatbot/views.py
from .models import Message

class MessageViewSet(viewsets.ModelViewSet):
    """消息视图集"""
    queryset = Message.objects.all()
    serializer_class = MessageSerializer
    permission_classes = [IsAuthenticated]
    
    def get_queryset(self):
        """获取当前用户的会话中的消息"""
        conversation_id = self.request.query_params.get('conversation_id')
        return self.queryset.filter(
            conversation__user=self.request.user,
            conversation_id=conversation_id
        )
    
    def create(self, request, *args, **kwargs):
        """创建消息"""
        conversation_id = request.data.get('conversation_id')
        role = request.data.get('role')
        content = request.data.get('content')
        
        # 验证会话
        try:
            conversation = Conversation.objects.get(
                id=conversation_id,
                user=request.user
            )
        except Conversation.DoesNotExist:
            return Response(
                {'error': '会话不存在'}, 
                status=status.HTTP_404_NOT_FOUND
            )
        
        # 创建消息
        message = Message.objects.create(
            conversation=conversation,
            role=role,
            content=content
        )
        
        serializer = self.get_serializer(message)
        return Response(serializer.data, status=status.HTTP_201_CREATED)
```

**实现说明**：
1. 验证会话是否属于当前用户
2. 创建消息记录
3. 返回消息数据

#### 3. 流式聊天实现

```python
# backend/chatbot/views.py
import requests
import json

@api_view(['POST'])
def stream_chat(request):
    """流式聊天接口"""
    conversation_id = request.data.get('conversation_id')
    message = request.data.get('message')
    
    # 获取会话
    conversation = Conversation.objects.get(
        id=conversation_id,
        user=request.user
    )
    
    # 获取用户API密钥
    profile = request.user.profile
    api_key = profile.openai_api_key
    
    # 调用OpenAI API
    headers = {
        'Authorization': f'Bearer {api_key}',
        'Content-Type': 'application/json'
    }
    
    data = {
        'model': conversation.model,
        'messages': [
            {'role': 'system', 'content': '你是一个智能助手'},
            {'role': 'user', 'content': message}
        ],
        'stream': True
    }
    
    # 发送请求
    response = requests.post(
        'https://api.openai.com/v1/chat/completions',
        headers=headers,
        json=data,
        stream=True
    )
    
    # 流式返回
    def event_stream():
        for line in response.iter_lines():
            if line:
                yield f'data: {line.decode("utf-8")}\n\n'
    
    return StreamingHttpResponse(
        event_stream(),
        content_type='text/event-stream'
    )
```

**实现说明**：
1. 接收用户消息
2. 调用OpenAI API
3. 使用 `StreamingHttpResponse` 实现流式响应
4. 实时返回AI回复

### 5.2.3 功能路由实现

功能路由模块实现了17种AI功能的智能路由。

```python
# backend/chatbot/function_router.py
import re

class FunctionRouter:
    """功能路由器"""
    
    def __init__(self):
        self.functions = {
            'chat': self._chat,
            'joke': self._joke,
            'story': self._story,
            'idiom': self._idiom,
            'translate': self._translate,
            'code': self._code,
            'weather': self._weather,
            'calculator': self._calculator,
            'poem': self._poem,
            'encyclopedia': self._encyclopedia,
            'news': self._news,
            'health': self._health,
            'finance': self._finance,
            'translate_en_zh': self._translate_en_zh,
            'translate_zh_en': self._translate_zh_en,
            'math': self._math,
            'search': self._search,
        }
    
    def route(self, intent, params):
        """路由到相应功能"""
        if intent in self.functions:
            return self.functions[intent](params)
        else:
            return self._chat(params)
    
    def _chat(self, params):
        """聊天功能"""
        return self._call_openai(params)
    
    def _joke(self, params):
        """笑话功能"""
        return "这是一个笑话：\n\n" + self._call_openai({
            'prompt': '讲一个笑话',
            **params
        })
    
    def _story(self, params):
        """故事功能"""
        return "这是一个故事：\n\n" + self._call_openai({
            'prompt': '讲一个故事',
            **params
        })
    
    def _translate(self, params):
        """翻译功能"""
        text = params.get('text', '')
        target_lang = params.get('target_lang', '英文')
        return self._call_openai({
            'prompt': f'将以下文本翻译成{target_lang}：{text}',
            **params
        })
    
    def _code(self, params):
        """编程功能"""
        return self._call_openai({
            'prompt': f'请帮我编写代码：{params.get("code", "")}',
            **params
        })
    
    def _weather(self, params):
        """天气查询功能"""
        city = params.get('city', '北京')
        return f"正在查询 {city} 的天气..."
    
    def _calculator(self, params):
        """计算器功能"""
        expression = params.get('expression', '')
        try:
            result = eval(expression)
            return f"计算结果：{result}"
        except:
            return "计算出错，请检查表达式"
    
    def _poem(self, params):
        """诗词生成功能"""
        return self._call_openai({
            'prompt': '请帮我生成一首诗词',
            **params
        })
    
    def _encyclopedia(self, params):
        """百科查询功能"""
        query = params.get('query', '')
        return f"正在查询：{query}..."
    
    def _news(self, params):
        """新闻查询功能"""
        category = params.get('category', '新闻')
        return f"正在获取 {category} 新闻..."
    
    def _health(self, params):
        """健康咨询功能"""
        return self._call_openai({
            'prompt': '请提供健康建议',
            **params
        })
    
    def _finance(self, params):
        """金融咨询功能"""
        return self._call_openai({
            'prompt': '请提供金融咨询建议',
            **params
        })
    
    def _translate_en_zh(self, params):
        """英译中功能"""
        text = params.get('text', '')
        return self._call_openai({
            'prompt': f'将以下英文翻译成中文：{text}',
            **params
        })
    
    def _translate_zh_en(self, params):
        """中译英功能"""
        text = params.get('text', '')
        return self._call_openai({
            'prompt': f'将以下中文翻译成英文：{text}',
            **params
        })
    
    def _math(self, params):
        """数学计算功能"""
        expression = params.get('expression', '')
        try:
            result = eval(expression)
            return f"计算结果：{result}"
        except:
            return "计算出错，请检查表达式"
    
    def _search(self, params):
        """搜索功能"""
        query = params.get('query', '')
        return f"正在搜索：{query}..."
    
    def _call_openai(self, params):
        """调用OpenAI API"""
        prompt = params.get('prompt', '')
        api_key = params.get('api_key', '')
        
        if not api_key:
            return "错误：未配置API密钥"
        
        try:
            import openai
            openai.api_key = api_key
            
            response = openai.ChatCompletion.create(
                model='gpt-3.5-turbo',
                messages=[
                    {'role': 'system', 'content': '你是一个智能助手'},
                    {'role': 'user', 'content': prompt}
                ]
            )
            
            return response.choices[0].message.content
        except Exception as e:
            return f"错误：{str(e)}"
```

**实现说明**：
1. 定义17种功能处理器
2. 根据用户意图路由到相应功能
3. 调用相应的AI模型处理请求

### 5.2.4 API服务实现

API服务模块封装了6个AI模型的API调用。

```python
# backend/chatbot/enhanced_api.py
import requests
import json

class EnhancedAPI:
    """增强型API服务"""
    
    def __init__(self):
        self.providers = {
            'openai': {
                'endpoint': 'https://api.openai.com/v1/chat/completions',
                'key_env': 'OPENAI_API_KEY'
            },
            'deepseek': {
                'endpoint': 'https://api.deepseek.com/v1/chat/completions',
                'key_env': 'DEEPSEEK_API_KEY'
            },
            'qwen': {
                'endpoint': 'https://dashscope.aliyuncs.com/compatible-mode/v1/chat/completions',
                'key_env': 'QWEN_API_KEY'
            },
            'gemini': {
                'endpoint': 'https://generativelanguage.googleapis.com/v1beta/models/gemini-pro:generateContent',
                'key_env': 'GEMINI_API_KEY'
            },
            'kimi': {
                'endpoint': 'https://api.moonshot.cn/v1/chat/completions',
                'key_env': 'KIMI_API_KEY'
            },
            'doubao': {
                'endpoint': 'https://ark.cn-beijing.volces.com/api/v3/chat/completions',
                'key_env': 'DOUBAO_API_KEY'
            }
        }
    
    def call_api(self, provider, messages, api_key):
        """调用AI API"""
        if provider not in self.providers:
            return {'error': '不支持的AI提供商'}
        
        provider_config = self.providers[provider]
        headers = {
            'Authorization': f'Bearer {api_key}',
            'Content-Type': 'application/json'
        }
        
        data = {
            'model': self._get_model(provider),
            'messages': messages
        }
        
        try:
            response = requests.post(
                provider_config['endpoint'],
                headers=headers,
                json=data
            )
            response.raise_for_status()
            return response.json()
        except Exception as e:
            return {'error': str(e)}
    
    def _get_model(self, provider):
        """获取模型名称"""
        models = {
            'openai': 'gpt-3.5-turbo',
            'deepseek': 'deepseek-chat',
            'qwen': 'qwen-plus',
            'gemini': 'gemini-pro',
            'kimi': 'moonshot-v1-8k',
            'doubao': 'doubao-1-5-pro'
        }
        return models.get(provider, 'gpt-3.5-turbo')
    
    def auto_switch(self, providers, messages):
        """自动切换API"""
        for provider in providers:
            api_key = self._get_api_key(provider)
            if api_key:
                result = self.call_api(provider, messages, api_key)
                if 'error' not in result:
                    return result
        return {'error': '所有API调用失败'}
    
    def _get_api_key(self, provider):
        """获取API密钥"""
        import os
        env_key = self.providers[provider]['key_env']
        return os.getenv(env_key)
```

**实现说明**：
1. 支持6个AI模型提供商
2. 实现自动API切换
3. 统一的API调用接口

```javascript
import { defineStore } from 'pinia'

        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        conversation_id: conversationId,
        role: 'user',
        content: content
      })
    })
    return await response.json()
  }
  
  return {
    conversations,
    messages,
    currentConversation,
    getConversations,
    createConversation,
    getMessages,
    sendMessage
  }
```
